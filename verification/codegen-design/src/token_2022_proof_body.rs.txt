const ORACLE_BOUND: usize = 65536;
const HEX_RADIX: u32 = 16;

fn rows() -> std::collections::BTreeMap<&'static str, Vec<u8>> {
    include_str!("oracle.tsv")
        .lines()
        .map(|line| {
            let mut fields = line.split('\t');
            let key = fields.next().unwrap();
            let hex = fields.next().unwrap();
            let bytes = (0..hex.len())
                .step_by(2)
                .map(|index| u8::from_str_radix(&hex[index..index + 2], HEX_RADIX).unwrap())
                .collect();
            (key, bytes)
        })
        .collect()
}

fn check<T: core::fmt::Debug + PartialEq, C: Encoder<T> + Decoder<T>>(
    codec: C,
    value: T,
    key: &str,
    rows: &std::collections::BTreeMap<&str, Vec<u8>>,
) {
    let limits = solana_kit_core::codecs::Limits::new(ORACLE_BOUND, ORACLE_BOUND);
    let encoded = codec.encode(&value, limits.max_bytes).unwrap();
    let dart = &rows[key];

    check_oracle(&encoded, dart, key);
    assert_eq!(
        codec.decode(&encoded, limits).unwrap(),
        value,
        "{key} decode"
    );
    check_dispatch(dart, limits, key);
    assert!(
        codec.encode(&value, encoded.len() - 1).is_err(),
        "{key} byte budget"
    );
    check_truncation(&codec, &encoded, limits, key);
}

/// Compares complete wire bytes or the independently encoded extension payload.
fn check_oracle(encoded: &[u8], dart: &[u8], key: &str) {
    if key.starts_with("extension-") && !key.starts_with("extension-uninitialized-") {
        assert_eq!(&encoded[..2], &dart[..2], "{key} tag");
        assert_eq!(
            usize::from(u16::from_le_bytes([encoded[2], encoded[3]])),
            dart.len() - 2,
            "{key} declared payload length"
        );
        assert_eq!(&encoded[4..], &dart[2..], "{key} independent payload");
    } else {
        assert_eq!(encoded, dart, "{key}");
    }
}

/// Requires public instruction dispatch to preserve the full independent encoding.
fn check_dispatch(dart: &[u8], limits: solana_kit_core::codecs::Limits, key: &str) {
    if !key.starts_with("account-") && !key.starts_with("extension-") {
        let data = Token2022InstructionDataCodec.decode(dart, limits).unwrap();
        assert_eq!(
            Token2022InstructionDataCodec
                .encode(&data, limits.max_bytes)
                .unwrap(),
            dart,
            "{key} complete dispatch"
        );
    }
}

/// Checks every truncated extension and the required instruction header boundary.
fn check_truncation<T, C: Decoder<T>>(
    codec: &C,
    encoded: &[u8],
    limits: solana_kit_core::codecs::Limits,
    key: &str,
) {
    if key.starts_with("extension-") {
        for end in 0..encoded.len() {
            assert!(
                codec.decode(&encoded[..end], limits).is_err(),
                "{key} truncation {end}"
            );
        }
    } else if !key.starts_with("account-") {
        assert!(codec.decode(&[], limits).is_err(), "{key} missing header");
    }
}
